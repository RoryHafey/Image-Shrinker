@echo off
echo Copying JDK to C:\Program Files\Java...

:: Create the target folder if it doesn't exist
if not exist "C:\Program Files\Java" (
    mkdir "C:\Program Files\Java"
)

:: Copy jdk-21 folder from D: to C:\Program Files\Java (mirror mode)
xcopy "D:\jdk-21" "C:\Program Files\Java\jdk-21" /E /I /H /Y

echo Setting JAVA_HOME and updating PATH...

:: Set JAVA_HOME
setx JAVA_HOME "C:\Program Files\Java\jdk-21"

:: Update PATH
setx PATH "%PATH%;%JAVA_HOME%\bin"

echo Done. Restart your terminal or system for changes to take effect.
exit
